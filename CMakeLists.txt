cmake_minimum_required(VERSION 3.8)
project(avatar_challenge)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)

set(dependencies
  rclcpp
  ament_index_cpp
  std_msgs
  geometry_msgs
  visualization_msgs
  moveit_core
  moveit_ros_planning
  moveit_ros_planning_interface
  moveit_msgs
  tf2
  tf2_eigen
  tf2_ros
  tf2_geometry_msgs
)

# The shapes a message can carry are the same ones the YAML can describe, so the
# .msg files mirror config/shapes.yaml one-for-one.
rosidl_generate_interfaces(${PROJECT_NAME}
  msg/Point2D.msg
  msg/ToolSpec.msg
  msg/SamplingSpec.msg
  msg/Segment.msg
  msg/Shape.msg
  msg/ShapeArray.msg
  DEPENDENCIES geometry_msgs
)

# Geometry, YAML parsing and redundancy resolution live in a library so they can
# be unit tested without a running move_group.
add_library(shape_tracing SHARED
  src/path_sampler.cpp
  src/yaml_loader.cpp
  src/shape_msg_conversion.cpp
  src/redundancy.cpp
  src/markers.cpp
  src/alphabet.cpp
  src/text_layout.cpp
  src/glyph_shapes.cpp
)
target_include_directories(shape_tracing PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>
)
target_link_libraries(shape_tracing yaml-cpp)
ament_target_dependencies(shape_tracing ${dependencies} Eigen3)

# The interfaces are generated by this same package, so they cannot be pulled in
# with ament_target_dependencies; link the generated typesupport target instead.
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(shape_tracing "${cpp_typesupport_target}")

add_executable(shape_tracer src/shape_tracer_node.cpp)
target_link_libraries(shape_tracer shape_tracing)
ament_target_dependencies(shape_tracer ${dependencies} Eigen3)

# Turns strings into letters for the tracer to draw. It publishes shape messages
# and nothing else, so it needs none of MoveIt.
add_executable(word_writer src/word_writer_node.cpp)
target_link_libraries(word_writer shape_tracing)
ament_target_dependencies(word_writer ${dependencies} Eigen3)

install(TARGETS shape_tracing
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin
)
install(TARGETS shape_tracer word_writer
  DESTINATION lib/${PROJECT_NAME}
)
install(DIRECTORY include/
  DESTINATION include
)
install(DIRECTORY launch config rviz
  DESTINATION share/${PROJECT_NAME}
)

# Development helpers: publish a shapes YAML on ~/add_shapes, send a word to
# word_writer, and render whatever is published into a PNG. See the file headers.
install(PROGRAMS
  scripts/publish_shapes.py
  scripts/publish_word.py
  scripts/preview_shapes.py
  DESTINATION lib/${PROJECT_NAME}
)

# Launch-time helper shared by start.launch.py and trace_shapes.launch.py.
ament_python_install_package(${PROJECT_NAME}_launch)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_shape_geometry test/test_shape_geometry.cpp)
  target_link_libraries(test_shape_geometry shape_tracing)
  ament_target_dependencies(test_shape_geometry ${dependencies} Eigen3)

  ament_add_gtest(test_shape_msg_conversion test/test_shape_msg_conversion.cpp)
  target_link_libraries(test_shape_msg_conversion shape_tracing "${cpp_typesupport_target}")
  ament_target_dependencies(test_shape_msg_conversion ${dependencies} Eigen3)

  # Reads the real font rather than a fixture: the shipped alphabet passing the
  # loader's geometry checks is exactly what is worth testing.
  ament_add_gtest(test_alphabet test/test_alphabet.cpp)
  target_compile_definitions(test_alphabet
    PRIVATE ALPHABET_YAML="${CMAKE_CURRENT_SOURCE_DIR}/config/alphabet.yaml")
  target_link_libraries(test_alphabet shape_tracing "${cpp_typesupport_target}")
  ament_target_dependencies(test_alphabet ${dependencies} Eigen3)
endif()

ament_export_include_directories(include)
ament_export_libraries(shape_tracing)
ament_export_dependencies(${dependencies} Eigen3 rosidl_default_runtime)
ament_package()
